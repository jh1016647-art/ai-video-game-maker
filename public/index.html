const express = require('express'); const path = require('path'); const dotenv = require('dotenv'); const OpenAI = require('openai'); dotenv.config(); const app = express(); const PORT = process.env.PORT || 3000; const openai = process.env.OPENAI_API_KEY ? new OpenAI({ apiKey: process.env.OPENAI_API_KEY }) : null; app.use(express.static(path.join(__dirname, 'public'))); app.use(express.json()); function titleCase(str = '') { return str .toLowerCase() .split(/\s+/) .filter(Boolean) .map((word) => word.charAt(0).toUpperCase() + word.slice(1)) .join(' '); } function sentenceCase(str = '') { return str .toLowerCase() .replace(/(^\s*\w|[.!?]\s*\w)/g, (match) => match.toUpperCase()); } function buildMockVideo(prompt = 'cyberpunk dragon rescue', style = 'cinematic') { const cleaned = sentenceCase(prompt || 'cyberpunk dragon rescue'); const palettes = { cinematic: 'Neon Cyan & Midnight Blue', anime: 'Electric Purple & Silver', realistic: 'Sunset Gold & Crimson', dreamy: 'Forest Emerald & Lavender', retro: 'Pixel Coral & Aqua' }; return { title: `${titleCase(prompt)} — ${style} storyboard`, concept: `A ${style} visual scene centered on ${cleaned}, blending grand spectacle, emotional focus, and a strong payoff reveal.`, duration: '45 seconds', style, palette: palettes[style] || palettes.cinematic, shots: [ `Wide establishing shot of ${cleaned} from a dramatic angle`, 'Hero enters frame with a purposeful movement beat', 'Mid-shot revealing the emotional stakes and key action', 'Visual payoff using layered effects and dynamic color contrast', 'Closing reveal with slow motion and atmospheric sound design' ], hook: `Open immediately on the most iconic image from ${cleaned} to capture the audience before the conflict escalates.` }; } function buildMockGame(prompt = 'cyberpunk dragon rescue', genre = 'action') { const cleaned = sentenceCase(prompt || 'cyberpunk dragon rescue'); const genres = { action: 'Action Adventure', puzzle: 'Puzzle Platformer', racing: 'High-Speed Racing', roguelike: 'Rogue-lite Survival' }; return { title: `${titleCase(prompt)} Quest`, genre: genres[genre] || 'Action Adventure', objective: `Guide the player through ${cleaned} by surviving hazards, unlocking upgrades, and defeating the final challenge.`, mechanics: [ 'Fast movement and responsive jump controls', 'Collectible power-ups that transform traversal and combat', 'Dynamic enemy patterns and environmental hazards', 'Short level loops with escalating challenge and rewards' ], levelIdea: `A vertical arena inspired by ${cleaned}, complete with hidden routes, collectible boosts, and a final boss arena.` }; } async function generateWithAI(prompt, type, fallback) { if (!openai) { return fallback; } try { const response = await openai.chat.completions.create({ model: process.env.OPENAI_MODEL || 'gpt-4o-mini', temperature: 0.8, messages: [ { role: 'system', content: `You are a creative director for a game and video concept studio. Return valid JSON for ${type}.` }, { role: 'user', content: prompt } ] }); const text = response.choices?.[0]?.message?.content || '{}'; const parsed = JSON.parse(text); return parsed; } catch (error) { console.error(`AI generation failed for ${type}:`, error.message); return fallback; } } async function createVideoConcept(prompt, style) { const fallback = buildMockVideo(prompt, style); const aiPrompt = `Create a creative video storyboard JSON with fields: title, concept, duration, style, palette, shots (array), hook. Prompt: ${prompt}. Style: ${style}.`; const result = await generateWithAI(aiPrompt, 'video', fallback); return { ...fallback, ...result, shots: Array.isArray(result.shots) ? result.shots : fallback.shots, title: result.title || fallback.title, concept: result.concept || fallback.concept, style: result.style || style, palette: result.palette || fallback.palette, hook: result.hook || fallback.hook }; } async function createGameConcept(prompt, genre) { const fallback = buildMockGame(prompt, genre); const aiPrompt = `Create a game design JSON with fields: title, genre, objective, mechanics (array), levelIdea. Prompt: ${prompt}. Genre: ${genre}.`; const result = await generateWithAI(aiPrompt, 'game', fallback); return { ...fallback, ...result, title: result.title || fallback.title, genre: result.genre || fallback.genre, objective: result.objective || fallback.objective, mechanics: Array.isArray(result.mechanics) ? result.mechanics : fallback.mechanics, levelIdea: result.levelIdea || fallback.levelIdea }; } app.get('/api/health', (_req, res) => { res.json({ status: 'ok', name: 'AI Video and Game Maker', aiEnabled: Boolean(openai) }); }); app.get('/api/generate', async (req, res) => { const prompt = req.query.prompt || 'cyberpunk dragon rescue'; const style = req.query.style || 'cinematic'; const genre = req.query.genre || 'action'; const [video, game] = await Promise.all([ createVideoConcept(prompt, style), createGameConcept(prompt, genre) ]); res.json({ prompt, style, genre, video, game, generatedAt: new Date().toISOString() }); }); app.get('/api/video', async (req, res) => { const prompt = req.query.prompt || 'future city skyline chase'; const style = req.query.style || 'cinematic'; const video = await createVideoConcept(prompt, style); res.json({ video }); }); app.get('/api/game', async (req, res) => { const prompt = req.query.prompt || 'floating atlas run'; const genre = req.query.genre || 'action'; const game = await createGameConcept(prompt, genre); res.json({ game }); }); app.get('/api/export', (req, res) => { const project = { prompt: req.query.prompt || 'creative project', style: req.query.style || 'cinematic', genre: req.query.genre || 'action', exportedAt: new Date().toISOString() }; res.setHeader('Content-Disposition', 'attachment; filename="project-export.json"'); res.json(project); }); app.get('*', (_req, res) => { res.sendFile(path.join(__dirname, 'public', 'index.html')); }); app.listen(PORT, () => { console.log(`AI Video and Game Maker running at http://localhost:${PORT}`); if (!openai) { console.log('No OPENAI_API_KEY detected. Using built-in creative mock generation for offline demos.'); } });
